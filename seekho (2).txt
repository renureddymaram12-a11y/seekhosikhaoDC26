function liveBatches(){
  const map=new Map();
  BATCHES.forEach(b=>map.set(b.id,{...b,students:[...b.students],promise:b.promise||PROMISE_SEED[b.id]}));
  S.myBatches.forEach(b=>map.set(b.id,{...b,students:[...b.students]}));
  S.joined.forEach(id=>{
    const b=map.get(id); if(!b) return;
    if(b.mentor==="me"||b.mentor===S.me.sampleOf) return;
    if(b.students.includes("me")||(S.me.sampleOf&&b.students.includes(S.me.sampleOf))) return;
    if(b.students.length<b.max) b.students.push("me");
  });
  return [...map.values()].map(b=>{
    let status=b.status;
    if(S.completed.includes(b.id)) status="completed";
    else if(status!=="completed"&&b.students.length>=MIN) status="running";
    return {...b,status,seats:b.students.length};
  });
}

window.joinBatch=function(id){
  if(!S.me.registered||!S.me.parentConsent){ toast("Register with parent consent first."); go("/register"); return; }
  if(S.joined.includes(id)){ toast("You are already in this batch."); return; }
  S.joined.push(id); save();
  const b=liveBatches().find(x=>x.id===id);
  if(b&&b.seats===MIN){ window.unlockFlash=id; toast(t("meet.unlocked")); } else toast(b&&b.seats>MIN?"You are on the list.":"You are on the list.");
  render();
};
window.completeBatch=function(id){
  const b=liveBatches().find(x=>x.id===id);
  if(!b) return;
  if(b.seats<MIN){ toast("Need 8 students before you close."); return; }
  const isMentor=b.mentor==="me"||(S.me.sampleOf&&b.mentor===S.me.sampleOf);
  if(!isMentor){ toast("Only the mentor can close a batch."); return; }
  if(S.completed.includes(id)||b.status==="completed"){ toast("Already closed."); return; }
  S.completed.push(id);
  S.myCerts.unshift({id:"SS-ME-"+id.slice(-6).toUpperCase(),user:"me",batch:id,role:"mentor",name:S.me.name,when:new Date().toLocaleDateString("en-IN")});
  const tag=id.slice(-6).toUpperCase(), when=new Date().toLocaleDateString("en-IN");
  b.students.forEach((sid,i)=>{
    const nm=sid==="me"?S.me.name:(person(sid)?.name||sid);
    S.myCerts.push({id:"SS-"+tag+"-S"+(i+1),user:sid,batch:id,role:"student",name:nm,when});
  });
  S.me.followers+=15; save(); toast("Batch closed. Mentor and all "+b.students.length+" students now have e-certificates."); render();
};

  const schedule=ENG_DAYS[+f.day.value]+"s "+fmtRange(startMin,startMin+mins)+" IST";
